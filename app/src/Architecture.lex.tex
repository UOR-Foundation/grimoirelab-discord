definition{
  name: edgesKnown,
  parameters: [count : Nat, records : (List Foundation.Arch::Edge)],
  result: Bool,
  body: (match records
    | List.nil => true
    | List.cons edge rest => (((project edge fromIndex) <? count) && (((project edge toIndex) <? count) && (edgesKnown count rest)))),
  recursive_argument: records
}

definition{
  name: viewpointsKnown,

    stakeholderCount : Nat,
    concernCount : Nat,
    records : (List Foundation.Arch::Viewpoint)
  ],
  result: Bool,
  body: (match records
    | List.nil => true
    | List.cons viewpoint rest => (((project viewpoint stakeholderIndex) <? stakeholderCount) && (((project viewpoint concernIndex) <? concernCount) && (((project viewpoint modelKindIndex) <? 3) && (viewpointsKnown stakeholderCount concernCount rest))))),
  recursive_argument: records
}

definition{
  name: viewMatches,
  parameters: [view : Foundation.Arch::View, viewpoints : (List Foundation.Arch::Viewpoint)],
  result: Bool,
  body: (match viewpoints
    | List.nil => false
    | List.cons viewpoint rest => ((((project viewpoint index) =? (project view viewpointIndex)) && ((project viewpoint modelKindIndex) =? (project view modelKindIndex))) || (viewMatches view rest))),
  recursive_argument: viewpoints
}

definition{
  name: viewsConform,
  parameters: [
    viewpoints : (List Foundation.Arch::Viewpoint),
    records : (List Foundation.Arch::View)
  ],
  result: Bool,
  body: (match records
    | List.nil => true
    | List.cons view rest => ((viewMatches view viewpoints) && (viewsConform viewpoints rest))),
  recursive_argument: records
}
